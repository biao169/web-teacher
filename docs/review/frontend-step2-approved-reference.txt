+"""Proposed public-only homepage reader. Reuses existing visibility/query rules."""
+async def homepage_rows(r,site):
+    teacher=await r.content.listing('profiles',public=True,query={
+        'size':10,'sort':'sort_order','direction':'asc','f.is_featured':1})
+    result={'profiles':teacher['rows'][:1]}
+    for table,field,default,sort in (
+        ('publications','homepage_publication_limit',6,'year'),
+        ('projects','homepage_project_limit',10,'start_date'),
+        ('news','homepage_news_limit',5,'published_at')):
+        value=site.get(field)
+        limit=max(0,int(default if value is None else value))
+        result[table]=[]
+        if not limit:continue
+        size=next((n for n in (10,20,50,100) if n>=limit),100)
+        page=1
+        while len(result[table])<limit:
+            batch=await r.content.listing(table,public=True,query={
+                'size':size,'page':page,'sort':sort,'direction':'desc','f.is_featured':1})
+            result[table].extend(batch['rows'][:limit-len(result[table])])
+            if page>=batch['pages'] or not batch['rows']:break
+            page+=1
+    return result
